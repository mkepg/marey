// Tick k sits at clock angle k * 6 degrees from 12 o'clock, i.e. k * 6 - 90 from +x.
// The tick is drawn 4 wide, 16 (or 32) tall, so its long side is vertical at rotation 0,
// which is radial at the top. Rotating by k * 6 keeps the long side along the radius.
// The tick's centre is placed on the radius-220 circle.
let ring = 220

scene {
  size: (800, 600)
  background: #0a0e1a

  generate k in 0 to 59 {
    let angle = k * 6 - 90
    rectangle tick {
      position: (400 + ring * cos(angle), 300 + ring * sin(angle))
      size: (4, if k % 5 == 0 then 32 else 16)
      rotation: k * 6
      color: white
    }
  }
}
