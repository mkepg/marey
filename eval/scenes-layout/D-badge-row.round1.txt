// Centres 100 apart, five badges: centres at 200, 300, 400, 500, 600.
// The badge is symmetric about its own (0, 0), so the row's bounding box is centred on 400.
template Badge() {
  circle disc {
    position: (0, 0)
    radius: 30
    color: cyan
  }
  rectangle pip {
    position: (0, 0)
    size: (20, 20)
    color: white
  }
}

scene {
  size: (800, 600)
  background: #0a0e1a

  generate i in 0 to 4 {
    use Badge() badge { position: (200 + i * 100, 300) }
  }
}
