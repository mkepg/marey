let count = 60
let ring  = 220

scene {
  size: (800, 600)
  background: #0a0e1a

  generate i in 0 to count - 1 {
    let angle = i * 360 / count - 90
    let major = i % 5 == 0
    // The rectangle's long side is vertical at rotation 0, which is already the
    // radial direction at 12 o'clock, so the rotation is the clockwise step from the top.
    rectangle tick {
      position: (400 + ring * cos(angle), 300 + ring * sin(angle))
      size: (4, if major then 32 else 16)
      rotation: i * 360 / count
      color: if major then orange else white
    }
  }
}
