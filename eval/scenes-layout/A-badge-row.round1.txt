template Badge() {
  circle disc { position: (0, 0), radius: 30, color: cyan }
  rectangle pip { position: (0, 0), size: (20, 20), color: #0a0e1a }
}

// Centres 100 apart, 4 steps = 400 wide, so they run 200..600.
scene {
  size: (800, 600)
  background: #0a0e1a

  generate i in 0 to 4 {
    use Badge() badge { position: (200 + i * 100, 300) }
  }
}
